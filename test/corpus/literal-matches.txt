======================
Literal match patterns
======================

func label(code:int) str {
    return match code {
        -1 => "negative"
        200 => "ok"
        else => "other"
    }
}

func enabled(value:bool) int {
    return match value {
        true => 1
        false => 0
    }
}

func command(value:str) int {
    return match value {
        "stop" => 1
        else => 0
    }
}
