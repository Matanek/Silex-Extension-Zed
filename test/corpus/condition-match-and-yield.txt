Condition match and nested yields

func category(score:int) str {
    return match {
        score >= 16 => "excellent"
        score >= 10 => "admitted"
        else => "insufficient"
    }
}

func predicted(condition:bool, x:float) float {
    return match condition {
        true => {
            var result = x * 10.0
            result /= 2.0
            yield match result > 5.0 {
                true => { yield result + 0.5 }
                else => result
            }
        }
        else => 0.0
    }
}
